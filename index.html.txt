    <script>
        const today = new Date().toISOString().split('T')[0];
        document.getElementById('date').value = today;
        document.getElementById('filterDate').value = today;

        let expenses = JSON.parse(localStorage.getItem('expenses_chosw') || '[]');

        function saveData() {
            localStorage.setItem('expenses_chosw', JSON.stringify(expenses));
            render();
        }

        document.getElementById('expenseForm').addEventListener('submit', function(e) {
            e.preventDefault();
            const date = document.getElementById('date').value;
            const amount = parseInt(document.getElementById('amount').value, 10);
            const usage = document.getElementById('usage').value.trim();
            const note = document.getElementById('note').value.trim();
            const editIndex = parseInt(document.getElementById('editIndex').value, 10);

            if (editIndex >= 0) {
                expenses[editIndex] = { date, amount, usage, note };
            } else {
                expenses.push({ date, amount, usage, note });
            }

            expenses.sort((a, b) => new Date(b.date) - new Date(a.date));
            resetForm();
            saveData();
        });

        function resetForm() {
            document.getElementById('expenseForm').reset();
            document.getElementById('date').value = today;
            document.getElementById('editIndex').value = "-1";
            document.getElementById('saveBtn').innerText = "+ 지출 내역 저장";
            document.getElementById('cancelBtn').style.display = "none";
        }

        function editExpense(index) {
            const item = expenses[index];
            document.getElementById('date').value = item.date;
            document.getElementById('amount').value = item.amount;
            document.getElementById('usage').value = item.usage;
            document.getElementById('note').value = item.note || '';
            document.getElementById('editIndex').value = index;
            document.getElementById('saveBtn').innerText = "수정 완료";
            document.getElementById('cancelBtn').style.display = "block";
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function deleteExpense(index) {
            if (confirm('이 지출 내역을 삭제하시겠습니까?')) {
                expenses.splice(index, 1);
                saveData();
            }
        }

        function render() {
            const filterDateVal = document.getElementById('filterDate').value;
            const filterMonthVal = document.getElementById('filterMonth').value;
            const listEl = document.getElementById('expenseList');
            listEl.innerHTML = '';

            const dailyTotals = {};
            let totalAll = 0;

            expenses.forEach(item => {
                dailyTotals[item.date] = (dailyTotals[item.date] || 0) + item.amount;
                totalAll += item.amount;
            });

            document.getElementById('grandTotal').innerText = totalAll.toLocaleString() + '원';

            let filtered = expenses.filter(item => {
                if (filterDateVal) return item.date === filterDateVal;
                if (filterMonthVal) return item.date.startsWith(filterMonthVal);
                return true;
            });

            let selectedDailySum = 0;
            if (filterDateVal && dailyTotals[filterDateVal]) {
                selectedDailySum = dailyTotals[filterDateVal];
            } else {
                selectedDailySum = filtered.reduce((acc, cur) => acc + cur.amount, 0);
            }
            document.getElementById('selectedDailyTotal').innerText = selectedDailySum.toLocaleString() + '원';

            if (filtered.length === 0) {
                listEl.innerHTML = '<div class="empty-msg">조건에 맞는 지출 내역이 없습니다.</div>';
                return;
            }

            filtered.forEach((item) => {
                const originalIndex = expenses.indexOf(item);
                const li = document.createElement('li');
                li.className = 'expense-item';
                li.innerHTML = `
                    <div class="expense-info">
                        <div class="date">${item.date}</div>
                        <div class="usage">${escapeHtml(item.usage)}</div>
                        ${item.note ? `<div class="note">${escapeHtml(item.note)}</div>` : ''}
                    </div>
                    <div class="expense-amount-wrap">
                        <div class="expense-amount">-${item.amount.toLocaleString()}원</div>
                        <div class="daily-cum">일일 누계: ${dailyTotals[item.date].toLocaleString()}원</div>
                    </div>
                    <div>
                        <button class="btn-delete" onclick="editExpense(${originalIndex})">✎</button>
                        <button class="btn-delete" onclick="deleteExpense(${originalIndex})">✕</button>
                    </div>
                `;
                listEl.appendChild(li);
            });
        }

        function escapeHtml(text) {
            return text.replace(/[&<>"']/g, function(m) {
                return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' }[m];
            });
        }

        document.getElementById('filterDate').addEventListener('change', function() {
            if(this.value) document.getElementById('filterMonth').value = '';
            render();
        });

        document.getElementById('filterMonth').addEventListener('change', function() {
            if(this.value) document.getElementById('filterDate').value = '';
            render();
        });

        function exportJSON() {
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(expenses, null, 2));
            const downloadAnchor = document.createElement('a');
            downloadAnchor.setAttribute("href", dataStr);
            downloadAnchor.setAttribute("download", `가계부_백업_${today}.json`);
            document.body.appendChild(downloadAnchor);
            downloadAnchor.click();
            downloadAnchor.remove();
        }

        function importJSON(event) {
            const file = event.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const data = JSON.parse(e.target.result);
                    if (Array.isArray(data)) {
                        expenses = data;
                        saveData();
                        alert('백업 데이터를 성공적으로 불러왔습니다.');
                    }
                } catch(err) {
                    alert('올바른 백업 파일(JSON)이 아닙니다.');
                }
            };
            reader.readAsText(file);
        }

        function exportData() {
            let csvContent = "\uFEFF날짜,금액,사용처,기타\n";
            expenses.forEach(e => {
                csvContent += `"${e.date}",${e.amount},"${e.usage.replace(/"/g, '""')}","${(e.note||'').replace(/"/g, '""')}"\n`;
            });
            const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = `지출내역_cho.s.w_${today}.csv`;
            link.click();
        }

        render();
    </script>
